#     <Job Zip="${TargetRoot}/CEVVideos.pak" Input="*.cevvideo;*.webm" Exclude="*.cryasset;.cevpreview/*.*" />

.cevpreview/*.*
*.tmp
*.$pak

# copies, and the WAV / Ogg exports in the asset folders) come from tools/list_unshipped_cev_files.py, which
# tools/pak_cev_assets.sh passes to rc.exe as /exclude_listfile=.
